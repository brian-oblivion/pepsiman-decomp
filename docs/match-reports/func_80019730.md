Unit `src/code_7d74.c`. Takes a slot for `key` from a SlotList: the first
free slot (key -1) among the used ones, else the next unused slot if the
list has capacity left (growing `count`), else NULL. The slot gets the key,
its flag word and `unk8` cleared, and its coordinate system (`unk4`)
reinitialised with `GsInitCoordinate2(NULL, ...)`.

```c
Slot *func_80019730(SlotList *list, s32 key) {
    Slot *s;
    s32 i;

    s = list->slots;
    for (i = 0; i < list->count; i++) {
        if (s->unkC == -1) {
            break;
        }
        s++;
    }
    if (i < list->count) {
        s->unkC = key;
        s->unk0 = 0;
        GsInitCoordinate2(NULL, (GsCOORDINATE2 *)s->unk4);
        s->unk8 = 0;
        return s;
    }
    if (i < list->capacity) {
        s = &list->slots[list->count];
        list->count++;
        s->unkC = key;
        s->unk0 = 0;
        GsInitCoordinate2(NULL, (GsCOORDINATE2 *)s->unk4);
        s->unk8 = 0;
        return s;
    }
    return NULL;
}
```
